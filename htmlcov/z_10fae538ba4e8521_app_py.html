116@app.post("/predict", response_model=BatchResponse, tags=["Prédiction"]) 

117def predict(data: PredictRequest, model=Depends(get_model)): 

118 """ 

119 Score(s) de défaut pour un ou plusieurs clients. 

120 

121 Accepte un identifiant unique ou une liste : 

122 - `{"client_ids": 100002}` 

123 - `{"client_ids": [100002, 100003, 100004]}` 

124 

125 - **predictions** : liste des scores pour les clients trouvés 

126 - **not_found** : liste des identifiants absents du dataset 

127 """ 

128 t0 = time.perf_counter() 

129 predictions = [] 

130 not_found = [] 

131 

132 for cid in data.client_ids: 

133 try: 

134 features = scoring_model.get_client_features(cid) 

135 except Exception as e: 

136 raise HTTPException(status_code=500, detail=f"Dataset lookup error client {cid}: {e}") 

137 

138 if features is None: 

139 not_found.append(cid) 

140 continue 

141 

142 try: 

143 result = scoring_model.predict(model, features) 

144 predictions.append(PredictionResponse(client_id=cid, **result)) 

145 except Exception as e: 

146 raise HTTPException(status_code=500, detail=f"Prediction error client {cid}: {e}") 

147 

148 scoring_model.update_stats(predictions, not_found) 

149 

150 _entry = { 

151 "endpoint": "/predict", 

152 "n_requested": len(data.client_ids), 

153 "n_scored": len(predictions), 

154 "n_not_found": len(not_found), 

155 "predictions": [ 

156 {"client_id": p.client_id, "score": p.score, "decision": p.decision} 

157 for p in predictions 

158 ], 

159 "not_found": not_found, 

160 "latency_ms": round((time.perf_counter() - t0) * 1000, 2), 

161 } 

162 logger.info(json.dumps(_entry)) 

163 _log_store.append(_entry) 

164 if len(_log_store) > _MAX_LOGS: 

165 _log_store.pop(0) 

166 

167 return BatchResponse(predictions=predictions, not_found=not_found)

170@app.post("/predict/new", response_model=PredictionResponse, tags=["Prédiction"]) 

171def predict_new(data: NewClientRequest, model=Depends(get_model)): 

172 """ 

173 Score de défaut pour un nouveau client sans historique dans le dataset. 

174 

175 Cas d'usage : demande de crédit en temps réel — le client n'est pas encore 

176 dans `train_processed_global.csv`. Les features pré-calculées en amont 

177 sont envoyées directement. L'exemple par défaut contient le **top 20 des 

178 features les plus importantes selon SHAP global** (+ colonnes one-hot des 

179 features catégorielles). Les 552 noms de features attendus sont disponibles 

180 via `GET /features`. 

181 

182 - Features manquantes → NaN, gérées nativement par LightGBM 

183 - **client_id** optionnel : numéro de dossier assigné par l'appelant 

184 """ 

185 t0 = time.perf_counter() 

186 try: 

187 result = scoring_model.predict(model, data.features) 

188 prediction = PredictionResponse(client_id=data.client_id, **result) 

189 scoring_model.update_stats([prediction], []) 

190 

191 _entry = { 

192 "endpoint": "/predict/new", 

193 "client_id": data.client_id, 

194 "score": prediction.score, 

195 "decision": prediction.decision, 

196 "n_features": len(data.features), 

197 "latency_ms": round((time.perf_counter() - t0) * 1000, 2), 

198 } 

199 logger.info(json.dumps(_entry)) 

200 _log_store.append(_entry) 

201 if len(_log_store) > _MAX_LOGS: 

202 _log_store.pop(0) 

203 

204 return prediction 

205 except Exception as e: 

206 raise HTTPException(status_code=500, detail=str(e))

209@app.get( 

210 "/predict/{client_id}/explain", 

211 response_model=ExplainResponse, 

212 tags=["Prédiction"], 

213) 

214def explain( 

215 client_id: int, 

216 top_n: int = Query(default=10, ge=1, le=50, description="Nombre de features à retourner"), 

217 model=Depends(get_model), 

218 explainer=Depends(get_explainer), 

219): 

220 """ 

221 Explication locale SHAP pour un client — pourquoi cette décision. 

222 

223 - **top_factors** : features classées par impact décroissant 

224 - `contribution` positive → pousse vers défaut (défavorable) 

225 - `contribution` négative → pousse vers remboursement (favorable) 

226 - **top_n** : nombre de features retournées (défaut 10, max 50) 

227 

228 Obligation réglementaire en crédit : tout refus doit être explicable. 

229 """ 

230 result = scoring_model.explain_client(model, explainer, client_id, top_n) 

231 

232 if result is None: 

233 raise HTTPException( 

234 status_code=404, 

235 detail=f"Client {client_id} introuvable dans le dataset.", 

236 ) 

237 

238 return ExplainResponse(**result)
